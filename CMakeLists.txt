cmake_minimum_required(VERSION 3.20)
project(jin_knapsack VERSION 0.1.0 LANGUAGES CXX)

option(KNAPSACK_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CheckCXXSourceCompiles)
check_cxx_source_compiles("int main() { __int128 x = 1; return int(x - 1); }" KNAPSACK_HAS_INT128)
if(NOT KNAPSACK_HAS_INT128)
  message(FATAL_ERROR "This implementation requires a C++20 compiler with __int128 (GCC or Clang).")
endif()

add_library(knapsack src/baseline.cpp src/hinted.cpp src/coloring.cpp src/solver.cpp)
target_include_directories(knapsack PUBLIC include PRIVATE src)
target_compile_options(knapsack PRIVATE -Wall -Wextra -Wconversion -Wshadow)
if(KNAPSACK_SANITIZERS)
  target_compile_options(knapsack PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(knapsack PUBLIC -fsanitize=address,undefined)
endif()

add_executable(knapsack_cli app/main.cpp)
target_link_libraries(knapsack_cli PRIVATE knapsack)
add_executable(knapsack_generate bench/generate.cpp)
target_include_directories(knapsack_generate PRIVATE app)
target_link_libraries(knapsack_generate PRIVATE knapsack)
add_executable(knapsack_bench bench/main.cpp)
target_include_directories(knapsack_bench PRIVATE app)
target_link_libraries(knapsack_bench PRIVATE knapsack)
add_executable(knapsack_extension_probe bench/extension_probe.cpp)
target_include_directories(knapsack_extension_probe PRIVATE app src)
target_link_libraries(knapsack_extension_probe PRIVATE knapsack)
target_compile_options(knapsack_extension_probe PRIVATE -Wall -Wextra -Wconversion -Wshadow)
add_executable(knapsack_multistar_probe bench/multistar_probe.cpp)
target_include_directories(knapsack_multistar_probe PRIVATE app src)
target_link_libraries(knapsack_multistar_probe PRIVATE knapsack)

include(CTest)
if(BUILD_TESTING)
  add_executable(knapsack_tests tests/test_main.cpp tests/components.cpp)
  target_include_directories(knapsack_tests PRIVATE src)
  target_link_libraries(knapsack_tests PRIVATE knapsack)
  target_compile_options(knapsack_tests PRIVATE -Wall -Wextra -Wconversion -Wshadow)
  add_test(NAME knapsack_correctness COMMAND knapsack_tests)
  add_test(NAME knapsack_cli_example COMMAND knapsack_cli --input ${CMAKE_CURRENT_SOURCE_DIR}/examples/small.txt --stats)
  set_tests_properties(knapsack_cli_example PROPERTIES PASS_REGULAR_EXPRESSION "profit 11")
  add_test(NAME knapsack_benchmark_smoke COMMAND knapsack_bench --n 30 --wmax 10 --verify)
  add_test(NAME knapsack_extension_probe_smoke COMMAND knapsack_extension_probe 4)
  add_test(NAME knapsack_graph_hint_contract COMMAND knapsack_extension_probe --graph 3)
  add_executable(knapsack_allocation_failure_test tests/allocation_failure.cpp)
  target_link_libraries(knapsack_allocation_failure_test PRIVATE knapsack)
  target_compile_options(knapsack_allocation_failure_test PRIVATE -Wall -Wextra -Wconversion -Wshadow)
  add_test(NAME knapsack_allocation_failure COMMAND knapsack_allocation_failure_test)
  add_test(NAME knapsack_multistar_contract COMMAND knapsack_multistar_probe 16)
endif()
